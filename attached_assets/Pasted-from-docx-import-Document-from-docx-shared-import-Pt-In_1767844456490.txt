# Helper to add headings
def h1(text):
    p = doc.add_paragraph(text)
    p.style = doc.styles['Heading 1']
    return p

    p = doc.add_paragraph(text)

def bullet(items, level=0):
    for it in items:
        p = doc.add_paragraph(it, style='List Bullet')
        if level:
            p.paragraph_format.left_indent = Inches(0.25*level)

h1("1) Product Summary")
doc.add_paragraph(

h1("7) Mechanics: Calculations (Simplified)")
h2("Rental core equations")
bullet([
    "Effective Rent = Expected Rent × (1 − VacancyRate)",
    "Monthly OpEx = (Taxes + Insurance)/12 + Maintenance + Management (+ utilities if used)",
    "NOI = Effective Rent − Monthly OpEx",
    "Cash Flow = NOI − Debt Service − (Holding Cost during rehab allocated monthly, optional)",
    "Cap Rate = (NOI × 12) / Purchase Price",
    "Cash-on-Cash = (Cash Flow × 12) / Cash Invested"
])

h2("Flip core equations")
bullet([
    "Total Cost = Purchase + Rehab + Holding + Financing costs + Selling costs",
    "Profit = Sale Price − Total Cost",
    "Sensitivity: ARV optimism increases downside variance; diligence reduces variance."
])
